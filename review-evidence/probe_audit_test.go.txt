package runner
import("testing";"errors";"os")
func TestAuditProbeHonorsRequestedStop(t *testing.T) {
 r,_,args := newProbeFixture(t)
 stop := NewStopController()
 r.AttachStopController(stop)
 stop.Request()
 _,err := r.Probe("fixture-model")
 var interrupted *InterruptedCallError
 if !errors.As(err,&interrupted) {
  _,statErr := os.Stat(args)
  t.Fatalf("stop ignored: error=%v child_started=%v",err,statErr==nil)
 }
 if _,statErr := os.Stat(args); !errors.Is(statErr,os.ErrNotExist) {
  t.Fatalf("pre-stopped probe started child: stat=%v",statErr)
 }
}
